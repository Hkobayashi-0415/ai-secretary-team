export function apiBase() {
  // Prefer window-injected base, then Vite env, then relative ('')
  const w = (window as any);
  let base = (w.API_BASE ?? import.meta.env.VITE_API_URL ?? '').toString();
  if (!base) return '';
  // When nginx provides /api proxy, our callers already append "/api/v1/...".
  // Avoid duplicating "/api" → return '' so requests become "/api/v1/...".
  if (base === '/api' || base === '/api/') return '';
  // Normalize trailing slash
  return base.replace(/\/$/, '');
}

export async function getJSON<T>(url: string, init?: RequestInit): Promise<T> {
  const r = await fetch(url, init);
  if (!r.ok) throw new Error(`HTTP ${r.status}: ${await r.text().catch(()=>'<no body>')}`);
  return r.json() as Promise<T>;
}

export type Assistant = { id: string; name: string };
export type Conversation = { id: string; title?: string | null };
export type Message = { id: string; role: 'user'|'assistant'|'system'; content: string };
export type MessagePage = { messages: Message[]; has_more: boolean };

export async function listAssistants(): Promise<Assistant[]> {
  const base = apiBase();
  return getJSON<Assistant[]>(`${base}/api/v1/assistants/`);
}

export async function listConversations(): Promise<Conversation[]> {
  const base = apiBase();
  return getJSON<Conversation[]>(`${base}/api/v1/conversations/`);
}

export async function createConversation(assistantId: string, userId?: string | null, title?: string | null): Promise<Conversation> {
  const base = apiBase();
  const body: any = { assistant_id: assistantId };
  if (userId) body.user_id = userId;
  if (title) body.title = title;
  // NOTE: FastAPI側はプレフィックス+"/"のパスを厳密に持つため末尾スラッシュを付与
  return getJSON<Conversation>(`${base}/api/v1/conversations/`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  });
}

export async function deleteConversation(id: string): Promise<void> {
  const base = apiBase();
  const r = await fetch(`${base}/api/v1/conversations/${id}`, { method: 'DELETE' });
  if (!r.ok) throw new Error(`HTTP ${r.status}`);
}

export async function listMessages(conversationId: string): Promise<Message[]> {
  const base = apiBase();
  return getJSON<Message[]>(`${base}/api/v1/conversations/${conversationId}/messages`);
}

export async function listMessagesPaged(conversationId: string, beforeId?: string, limit = 20): Promise<MessagePage> {
  const base = apiBase();
  const params = new URLSearchParams();
  if (beforeId) params.set('before_id', beforeId);
  if (limit) params.set('limit', String(limit));
  const qs = params.toString();
  const url = `${base}/api/v1/conversations/${conversationId}/messages/page${qs ? `?${qs}` : ''}`;
  return getJSON<MessagePage>(url);
}

export async function patchConversation(id: string, patch: { title?: string | null }): Promise<Conversation> {
  const base = apiBase();
  return getJSON<Conversation>(`${base}/api/v1/conversations/${id}`, {
    method: 'PATCH',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(patch),
  });
}
